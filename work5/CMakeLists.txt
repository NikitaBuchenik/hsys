cmake_minimum_required(VERSION 3.24)
project(work5_reduction LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
find_package(CUDAToolkit REQUIRED)
include(FetchContent)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.15.2 GIT_SHALLOW TRUE)
FetchContent_Declare(googlebenchmark GIT_REPOSITORY https://github.com/google/benchmark.git GIT_TAG v1.9.1 GIT_SHALLOW TRUE)
FetchContent_Declare(eigen GIT_REPOSITORY https://gitlab.com/libeigen/eigen.git GIT_TAG 3.4.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(googletest googlebenchmark)
FetchContent_GetProperties(eigen)
if(NOT eigen_POPULATED)
  if(POLICY CMP0169)
    cmake_policy(SET CMP0169 OLD)
  endif()
  FetchContent_Populate(eigen)
endif()
add_library(reduction_core core/src/reduction.cu)
target_include_directories(reduction_core PUBLIC core/include)
target_link_libraries(reduction_core PUBLIC CUDA::cudart)
# Host-only tests keep Eigen out of nvcc's device compilation.
add_executable(vector_tests tests/vector_tests.cpp)
target_include_directories(vector_tests SYSTEM PRIVATE ${eigen_SOURCE_DIR})
target_compile_definitions(vector_tests PRIVATE EIGEN_DONT_PARALLELIZE)
target_link_libraries(vector_tests PRIVATE reduction_core GTest::gtest_main)
add_executable(vector_benchmark benchmarks/vector_benchmark.cpp)
target_link_libraries(vector_benchmark PRIVATE reduction_core benchmark::benchmark)
enable_testing()
add_test(NAME vector_tests COMMAND vector_tests)
