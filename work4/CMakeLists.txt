cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "RTX 2060")
endif()
project(work4_wmma LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
include(FetchContent)
find_package(CUDAToolkit REQUIRED)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.15.2 GIT_SHALLOW TRUE)
FetchContent_Declare(googlebenchmark GIT_REPOSITORY https://github.com/google/benchmark.git GIT_TAG v1.9.1 GIT_SHALLOW TRUE)
FetchContent_Declare(eigen GIT_REPOSITORY https://gitlab.com/libeigen/eigen.git GIT_TAG 3.4.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(googletest googlebenchmark)
FetchContent_GetProperties(eigen)
if(NOT eigen_POPULATED)
  if(POLICY CMP0169)
    cmake_policy(SET CMP0169 OLD)
  endif()
  FetchContent_Populate(eigen)
endif()
add_library(eigen_headers INTERFACE)
target_include_directories(eigen_headers SYSTEM INTERFACE ${eigen_SOURCE_DIR})
target_compile_definitions(eigen_headers INTERFACE EIGEN_DONT_PARALLELIZE)
add_library(matrix_core core/src/matrix.cu)
target_include_directories(matrix_core PUBLIC core/include)
target_link_libraries(matrix_core PUBLIC CUDA::cudart)
add_executable(matrix_tests tests/matrix_tests.cu)
target_link_libraries(matrix_tests PRIVATE matrix_core eigen_headers GTest::gtest_main)
# Eigen is a CPU reference only; do not compile its math helpers for CUDA.
target_compile_definitions(matrix_tests PRIVATE EIGEN_NO_CUDA EIGEN_DONT_VECTORIZE)
add_executable(matrix_benchmark benchmarks/matrix_benchmark.cu)
target_link_libraries(matrix_benchmark PRIVATE matrix_core benchmark::benchmark)
enable_testing()
add_test(NAME matrix_tests COMMAND matrix_tests)
